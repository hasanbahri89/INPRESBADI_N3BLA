// Kode Google Apps Script Real-Time untuk SMP Negeri 3 Blanakan
// Buat Sheet: "Pengawas", "Peserta", "Kehadiran", "Pengaturan"

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'getAll';
  
  if (action === 'getAll') {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      pengawas: getSheetData(ss, 'Pengawas'),
      peserta: getSheetData(ss, 'Peserta'),
      kehadiran: getSheetData(ss, 'Kehadiran'),
      pengaturan: getSheetData(ss, 'Pengaturan')
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const payload = JSON.parse(e.postData.contents);
  
  if (payload.action === 'addKehadiran') {
    const sheet = ss.getSheetByName('Kehadiran') || ss.insertSheet('Kehadiran');
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['id', 'tanggal', 'waktuScan', 'nomorPeserta', 'nama', 'jenisKelamin', 'nis', 'nisn', 'kelas', 'ruang', 'mapel', 'jamMulai', 'jamBerakhir', 'kodePengawas', 'namaPengawas', 'nipPengawas']);
    }
    const d = payload.data;
    sheet.appendRow([d.id, d.tanggal, d.waktuScan, d.nomorPeserta, d.nama, d.jenisKelamin || 'L', d.nis, d.nisn, d.kelas, d.ruang, d.mapel, d.jamMulai, d.jamBerakhir, d.kodePengawas, d.namaPengawas, d.nipPengawas || '-']);
  } else if (payload.action === 'syncPeserta') {
    const sheet = ss.getSheetByName('Peserta') || ss.insertSheet('Peserta');
    sheet.clearContents();
    sheet.appendRow(['nomorPeserta', 'nama', 'jenisKelamin', 'nis', 'nisn', 'kelas', 'ruang']);
    const list = payload.data || [];
    if (list.length > 0) {
      const rows = list.map(p => [p.nomorPeserta, p.nama, p.jenisKelamin || 'L', p.nis, p.nisn, p.kelas, p.ruang]);
      sheet.getRange(2, 1, rows.length, 7).setValues(rows);
    }
  } else if (payload.action === 'syncPengawas') {
    const sheet = ss.getSheetByName('Pengawas') || ss.insertSheet('Pengawas');
    sheet.clearContents();
    sheet.appendRow(['kodePengawas', 'nama', 'nip', 'tugasJabatan', 'password', 'role']);
    const list = payload.data || [];
    if (list.length > 0) {
      const rows = list.map(u => [u.kodePengawas, u.nama, u.nip, u.tugasJabatan, u.password, u.role]);
      sheet.getRange(2, 1, rows.length, 6).setValues(rows);
    }
  } else if (payload.action === 'syncPengaturan') {
    const sheet = ss.getSheetByName('Pengaturan') || ss.insertSheet('Pengaturan');
    sheet.clearContents();
    sheet.appendRow(['key', 'value']);
    const cfg = payload.data || {};
    sheet.appendRow(['schoolLogoUrl', cfg.schoolLogoUrl || '']);
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getSheetData(ss, name) {
  const sheet = ss.getSheetByName(name);
  if (!sheet || sheet.getLastRow() < 2) return [];
  const rows = sheet.getDataRange().getValues();
  const headers = rows.shift();
  return rows.map(r => {
    let obj = {};
    headers.forEach((h, i) => obj[h] = String(r[i] !== undefined && r[i] !== null ? r[i] : ''));
    return obj;
  });
}
